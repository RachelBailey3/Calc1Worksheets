\begin{question}
    When will the ball hit the ground?
     
        \begin{explanation}
            
       
            To determine when the ball hits the ground we need to solve the equation
            \[ s(t)=0 \]
            for $t$.  That is,
            \begin{align*}
            40t-16t^2 &= 0\\
            \answer{t(40-16t)} &= 0 \text{ (factor)}.
            \end{align*}
            This  equation has two solutions:
            \[t=\answer{0} \text{ and } t=\answer{2.5}
            \]
            seconds.
              
               Since the ball hits
            the ground  some time after it's
            thrown, we conclude that the ball hits the ground at time when $t>0$.
         
            The ball will hit ground at the time
            \[ t=\answer{2.5} \text{ seconds.} \]
        \end{explanation}
 
\end{question}

\begin{example}
In our previous example, we computed \textit{average velocity} on
several different intervals. We know that if we let $t\to 2$, the size of the interval will go to $0$ thus, we are computing the average velocity on smaller and
smaller time intervals. Recall that the limit of these average velocities is the \emph{instantaneous velocity} at $t=2$.
\begin{question}

    Find $v(2)$, the instantaneous velocity of the ball $2$ seconds after it is thrown.
     
    \begin{explanation}
        From the previous example, we know that the average velocity of the
        ball on the interval $[t,2]$, for $0<t<2$, and the average velocity
        on the interval $[2,t]$, for $2<t<2.5$, are both given by
        \begin{align*}
        v_{\text{av}}(t) &=  \frac{s(t)-s(2)}{t-2}\\ 
                &= \frac{\answer{(40t-16t^2)}-\answer{16} }{\answer{t-2}}\\
                &= \frac{-8\answer{(2t^2-5t+2)}}{t-2}\\
                &= \frac{8(2t-1)\answer{(t-2)}}{-\answer{(t-2)}}\\
                &= \answer{-8(2t-1)}\\
                &= \answer{8-16t}\hspace{0.06in}\text{ ft/s}.
        \end{align*}
       
        
        In order to find the instantaneous velocity $v(2)$, we
        take the limit as $t$ goes to $\answer{2}$.
        \begin{align*}
            v (2) &= \lim_{t\to2}v_{\text{av}}(t)\\
                &=\lim_{t\to2}  \frac{s(t)-s(2)}{t-2} \\
                &= \lim_{t\to2}(8-16t)\\
                &= \answer{-24}\text{ ft/s}.
        \end{align*}
    \end{explanation}

 
   
     
 
  
\end{question}
 
 
\end{example}
