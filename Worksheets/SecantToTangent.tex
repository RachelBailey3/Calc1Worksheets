\begin{question} 
\begin{enumerate}
    \item Suppose a car traveled 100 miles in 2 hours. What was the car's average speed during this time?
 
  \item Now suppose we know that the car had traveled 57 miles after 1 hour and 113 miles after 3 hours. What was the car's average speed during the last two hours?

  \item If the car had traveled 35  miles after 1 hour and 80 miles after 2 hours, what was it's average speed between the first hour and second hour?
 
\end{enumerate}

   
\end{question}
